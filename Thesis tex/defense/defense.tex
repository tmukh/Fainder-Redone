% ---------------------------------------------------------------- 1. Context
\section{Problem}

\begin{frame}{Distribution-Aware Dataset Search and Fainder}
  \begin{itemize}
    \item Raw data not searchable (privacy, licensing, scale) $\to$ per-column \textbf{histograms}
    \item Predicate class: \emph{``$p$-th percentile $\bowtie$ threshold $\tau$''}
    \item \textbf{Fainder} [Behme et al., VLDB 2024]:
    \begin{itemize}
      \item K-means clusters $\to$ shared bin grid per cluster $\to$ precomputed cumulative densities
      \item Percentile predicate $\Rightarrow$ \textbf{binary search}
    \end{itemize}
    \item GitTables, 5M histograms: $>$2 orders of magnitude vs profile scan, 1\,TB $\to$ 3.2\,GB
  \end{itemize}
  \vfill
  \begin{alertblock}{This thesis}
    How does Fainder's query phase run on a modern multi-core server, and what makes it fast?
  \end{alertblock}
\end{frame}

\begin{frame}{Fainder in Plain Words}
  \begin{center}
  \resizebox{\linewidth}{!}{%
  \begin{tikzpicture}[
      node distance=4mm,
      box/.style={draw, rounded corners=1pt, align=center, font=\footnotesize, inner sep=5pt, minimum height=11mm},
      arr/.style={-{Stealth[length=2mm]}, thick}
    ]
    \node[box] (a) {5{,}017{,}619 column\\summaries (histograms)};
    \node[box, right=of a] (b) {grouped into $\sim$191\\families of similar shape};
    \node[box, right=of b] (c) {one shared x-axis\\per family};
    \node[box, right=of c] (d) {at each x-axis point:\\sorted fractions $+$ IDs};
    \node[box, right=of d] (e) {a query jumps into\\two sorted lists};
    \draw[arr] (a) -- (b);
    \draw[arr] (b) -- (c);
    \draw[arr] (c) -- (d);
    \draw[arr] (d) -- (e);
  \end{tikzpicture}}
  \end{center}
  \begin{itemize}
    \item Histogram $=$ ``how many values fall in each range''
    \item One family, one x-axis: the same question lands at the same position everywhere
    \item Stored per x-axis point: the fraction of each column's values below it --- \textbf{sorted}, with IDs
    \item Sorted $=$ binary-searchable: $\sim$191 short lists, never 5M summaries
    \item \alert{``median $> 20$'' $\iff$ ``fraction of values $\leq 20$ is $< 0.5$''}: $\tau$ picks the list, $p$ picks the cut
    \item Clusters are independent $\to$ parallel. Search $=$ few dependent reads; emit $=$ many writes
  \end{itemize}
\end{frame}

\begin{frame}{The Python Baseline}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{itemize}
      \item GIL: one thread at a time; \texttt{searchsorted} never releases it
      \item Thread sweep \textbf{flat}: 2\% spread at any $t$
      \item AoS layout: unread ID next to every value
      \item Interpreter dispatch on every iteration
    \end{itemize}
    \column{0.42\textwidth}
    \begin{block}{10{,}000 queries, 5M histograms}
      \centering
      \Large $\sim$2.4 hours \\[0.5ex]
      \normalsize (8{,}589\,s, any thread count)
    \end{block}
  \end{columns}
  \vfill
  \begin{itemize}
    \item IPC 2.35 vs Rust's 0.96 --- yet $378\times$ slower: $6.2\times$ the instructions, one effective core
    \item Rust wins by executing fewer instructions across all cores in parallel, not by winning on IPC
  \end{itemize}
\end{frame}

\begin{frame}{Goal and Scope}
  \begin{itemize}
    \item Close the algorithm--implementation gap; query results unchanged
    \item Rust query engine behind the existing Python API (PyO3); $>$20 optimisation axes ablated
    \item Constraint: bitwise-identical results, precision $=$ recall $=$ 1.000, every build
    \item Out of scope: algorithm changes, GPU, distributed execution
  \end{itemize}
  \vfill
  \begin{alertblock}{The investigation is the thesis, the port is its instrument}
    Four hardware ceilings $+$ negative composability $+$ a pre-flight methodology.
  \end{alertblock}
\end{frame}

% ---------------------------------------------------------------- 3. Results
\section{Results: Four Ceilings}

\begin{frame}{Finding 1: Four Performance Ceilings}
  \begin{columns}[T]
    \column{0.60\textwidth}
    \includegraphics[width=\linewidth]{ch4_four_ceilings.pdf}
    \column{0.38\textwidth}
    \small
    \begin{enumerate}
      \item Single reads ($t \leq 8$): \bad{refuted}
      \item Shared L3 ($t \approx 16$)
      \item Memory-traffic pressure ($t \in [32,48]$)
      \item Cross-socket link ($t > 48$)
    \end{enumerate}
    \vfill
    One regime each.\\ One intervention class each.
  \end{columns}
\end{frame}

\begin{frame}{Ceiling (i): Single-Thread Latency, Refuted}
  What it is: binary search is a \textbf{dependent-load chain} --- probe $i{+}1$ cannot start until probe $i$ returns. The literature's default bottleneck for in-memory search.
  \begin{center}
    \small
    \begin{tabular}{llr}
      \toprule
      Ablation & Idea & vs default, all 18 cells \\
      \midrule
      \texttt{simd} & vectorise the last $\sim$16 steps & $0.90$--$1.07\times$ \; noise \\
      \texttt{batch-search} & overlap 8 searches' loads & $0.96$--$1.09\times$ \; noise \\
      \texttt{horizontal-simd} & 16 queries walk in lockstep & $0.95$--$1.12\times$ \; noise \\
      \texttt{pgm} & predict the position, skip the chain & $0.91$--$1.13\times$ \; noise \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item Counters at $t{=}1$: IPC $\approx 1.9$, L1d miss/load $0.5$--$0.7\%$ --- the chains are cache-resident
    \item PGM verifiably cuts chain depth to $O(\log\log n)$; the wall does not move
    \item The wall is the \emph{number} of chains (one per cluster), not the depth of any single chain
  \end{itemize}
\end{frame}

\begin{frame}{Ceiling (ii): Shared L3 Capacity}
  The claim: at $t \in [8,32]$ the threads' combined working sets overflow the shared 105\,MiB L3. Two probes ($<1$ = faster than default, primary workload):
  \begin{center}
    \footnotesize
    \begin{tabular}{lrrrrrr}
      \toprule
      Build & $t{=}1$ & $t{=}8$ & $t{=}16$ & $t{=}32$ & $t{=}64$ & $t{=}96$ \\
      \midrule
      \texttt{aos} (capacity probe: doubles bytes/line) & $1.03$ & \bad{$1.17$} & $1.02$ & $1.00$ & \good{$0.96$} & $0.99$ \\
      \texttt{pooled} (allocator probe: no per-cluster allocs) & \good{$0.53$} & \good{$0.66$} & \good{$0.80$} & $0.95$ & $0.93$ & $1.02$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item \texttt{aos} changes \emph{only} footprint; its penalty sits inside the window ($+17\%$ at $t{=}8$, $\sim$2$\times$ L3 working set) and vanishes outside $\to$ capacity is real in $[8,32]$
    \item Yet the biggest win in that window fixes the allocator's lock, not capacity --- and it fades ($0.66 \to 0.95$) as ceiling (iii) takes over
    \item \texttt{f16} (halves bytes/value): null on the dense primary, pays $\sim$20\% at the sparse OOD density --- footprint only matters where it actually binds
  \end{itemize}
  \smallskip
  Verdict: one regime, two mechanisms --- allocator queueing on dense clusters, L3 footprint on sparse.
\end{frame}

\begin{frame}{Ceiling (iii): On-Socket Memory-Traffic Pressure}
  What it is: all cores share the queues on the path to memory (L3 $\to$ mesh $\to$ memory controller). Each added core lengthens every queue, so \emph{every} access waits longer --- while total traffic sits at $\sim$0.5\% of STREAM peak. \textbf{Congestion, not saturation.}
  \vfill
  The signature, \texttt{c256\_30gb}, wall in seconds:
  \begin{center}
    \small
    \begin{tabular}{lrrr}
      \toprule
      Build & $t{=}64$ & $t{=}96$ & step \\
      \midrule
      default & $4.15$ & $5.89$ & \bad{$+42\%$} \quad (IPC $1.93 \to 0.55$, LLC misses $\times 1.6$) \\
      \texttt{packed-ids} & $3.88$ & $4.70$ & \bad{$+21\%$} \quad (same step, half the damage) \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item Anti-scaling: adding 32 cores makes the run $42\%$ \emph{slower}
    \item Fix class $=$ fewer bytes per operation: \texttt{packed-ids} emits 20-bit instead of 32-bit IDs ($-37\%$ emit bytes) $\to$ \good{$-20\%$} at the anti-scaling cell
  \end{itemize}
\end{frame}

\begin{frame}{Ceiling (iv): Cross-Socket UPI Interconnect}
  Same workload, four memory placements (\texttt{numactl} probe, wall in seconds):
  \begin{center}
    \small
    \begin{tabular}{llrrrr}
      \toprule
      Placement & meaning & $t{=}32$ & $t{=}48$ & $t{=}64$ & $t{=}96$ \\
      \midrule
      A: unpinned & OS default, first-touch & $19.28$ & $16.97$ & $18.19$ & $17.58$ \\
      B: single socket & cores $+$ memory on socket 0 & \good{$\mathbf{16.77}$} & \good{$\mathbf{15.04}$} & -- & -- \\
      C: interleave & pages alternate sockets per 4\,KB & \bad{$25.87$} & \bad{$23.73$} & \bad{$22.91$} & \bad{$24.83$} \\
      D: cross-socket & memory socket 0, cores socket 1 & $19.08$ & $16.86$ & -- & -- \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item B vs A: removing all cross-socket traffic saves \good{11--13\%} --- that saving \emph{is} the link cost
    \item C vs A: ``fair'' page balancing costs \bad{26--41\%}: half of all accesses land remote, and the per-page alternation defeats the prefetcher
    \item D $\approx$ A: even the worst case matches default $\to$ the baseline was already paying the link
  \end{itemize}
\end{frame}

% ---------------------------------------------------- 4. Structural finding
\section{Results: Negative Composability}

\begin{frame}{Finding 2: Optimisations Compose Negatively}
  Five pairs, five axes, one shape: \emph{the coarser optimisation absorbed the headroom; the finer one's overhead becomes net cost.}
  \begin{center}
    \small
    \begin{tabular}{llr}
      \toprule
      Composition & Intended axis & Outcome \\
      \midrule
      \bcombo{} $+$ \texttt{packed-ids} & emit-ID bandwidth & \bad{NEG} 12/18 cells, up to $+24\%$ \\
      \bcombo{} $+$ qbatch $K{=}128$ & cold-load amortisation & \bad{NEG} every cell, $+8$ to $+32\%$ \\
      qbatch $+$ morsel & cross-worker coordination & \bad{NEG} every cell, $+8$ to $+12\%$ \\
      \texttt{packed-ids} $+$ \texttt{local-ids} & read-side bandwidth & \bad{NEG} $+186$ to $+245\%$ \\
      first-touch $+$ interleave & memory-locality balance & \bad{NEG} every cell, $+26$ to $+41\%$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item Every pair had a plausible positive argument; every one falsified by \texttt{perf} counters
    \item Five axes, one shape: structural, not coincidence
  \end{itemize}
\end{frame}

\begin{frame}{Finding 3: Pre-Flight Ceiling Identification}
  Classical ablation (build $\to$ measure $\to$ keep) assumes optimisations are independent. Here they are not: what helps depends on the binding ceiling \emph{at the target composition}.
  \vfill
  Before writing code:
  \begin{enumerate}
    \item Probe the binding ceiling at the target composition (\texttt{perf})
    \item Does the ablation's axis still bind there? If not: stop
    \item Predict the saving (bandwidth-budget arithmetic)
    \item Measure where the ceiling binds; separate saving from overhead
    \item A negative composition counts as a result
  \end{enumerate}
  \vfill
  Applies to any hardware-near system with stacked ceilings.
\end{frame}

\begin{frame}{Dispatch on Regime}
  \begin{itemize}
    \item No dominating build $\to$ per-regime selection: $(n_\text{hists}, t) \to$ build $+$ flags
    \item \bcombo{} at moderate $t$ / big data; \texttt{packed-ids} at $t{=}64$; qbatch at $t{=}96$; columnar at low $t$; default on small data
  \end{itemize}
  \vfill
  Validation:
  \begin{itemize}
    \item In-distribution: \good{17/18} exact, \good{18/18} within 5\%
    \item Out-of-distribution ($3.2\times$ sparser, never tuned on): \good{6/6} within 1\%
    \item Combined: \good{23/24}
  \end{itemize}
  \vfill
  Boundaries are machine-specific; the probing procedure is not.
\end{frame}

\begin{frame}{Future Work}
  \begin{itemize}
    \item (F1) Transparent Huge Pages: $512\times$ fewer TLB misses per covered page $\to$ targets the local-ids mechanism
    \item (F2) NUMA-aware build for $t > 48$: shard clusters across sockets, node-local workers
    \item (F3) Adaptive precision: f32 at low $t$, f16 at high $t$ / sparse density, selected at runtime
    \item Column-centric: widen its $t{=}1$ win ($2.17$--$2.63\times$) by fixing cross-cluster imbalance
    \item Dispatch validation beyond the $\sim$8.2\,K--26.3\,K hists/cluster range
  \end{itemize}
\end{frame}

\begin{frame}{Closing Remark}
  \begin{itemize}
    \item The speedups are specific to this codebase, machine, and workload
    \item What generalises: whether optimisation $B$ helps depends on which ceiling $A$ already collected --- invisible to per-operator cost models
  \end{itemize}
  \vfill
  \begin{alertblock}{Closing remark}
    Identify the binding ceiling at the target composition before committing the implementation cost of the next optimisation.
  \end{alertblock}
\end{frame}

\section{Backup}

\begin{frame}{Backup: One Query, End to End}
  Query: \emph{``median $>$ 20''} $=$ $(p{=}0.5,\; >,\; \tau{=}20)$
  \hfill \alert{median $> 20 \iff$ CDF$(20) < 0.5$}
  \vspace{1ex}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \centering
    Cluster A index (toy: 3 histograms)\\[0.5ex]
    \footnotesize
    bin edges: $[0, 10, \mathbf{20}, 30, 50]$\\[1ex]
    \begin{tabular}{lll}
      \toprule
      edge & sorted CDFs & IDs \\
      \midrule
      $\leq$10 & $[0.00, 0.10, 0.20]$ & H2 H1 H5 \\
      $\boldsymbol{\leq}$\textbf{20} & $[\mathbf{0.25}, \mathbf{0.45}, 0.50]$ & \textbf{H2 H5} H1 \\
      $\leq$30 & $[0.75, 0.75, 0.80]$ & H2 H5 H1 \\
      \bottomrule
    \end{tabular}
    \column{0.50\textwidth}
    \begin{enumerate}
      \item Binary search $\tau{=}20$ in the bin edges $\to$ column ``$\leq$20''
      \item Binary search $0.5$ in that sorted column $\to$ cut after 2 entries
      \item Emit the IDs before the cut: \good{H2, H5} \\ {\footnotesize(H1 sits exactly at $0.50$: median $= 20$, not $> 20$)}
      \item Cluster B (salaries, range $\geq$ 10\,000): $\tau$ below range $\to$ \textbf{all match, no search}
      \item Union across clusters $=$ the answer
    \end{enumerate}
  \end{columns}
  \vfill
  Clusters are independent $\to$ the cluster loop parallelises. Search $=$ few dependent reads; emit $=$ many writes.
\end{frame}

\begin{frame}{Backup: Why is end-to-end $530\times$ if search is only $1.34\times$?}
  \begin{itemize}
    \item Python's per-(query, cluster) result-dict construction dominates its wall. Measured at $t{=}16$, materialisation accounts for:
    \begin{itemize}
      \item \texttt{c256\_10gb}: $580.9$\,s of $599.7$\,s ($96.9\%$)
      \item \texttt{c256\_30gb}: $1{,}878.5$\,s of $1{,}890.3$\,s ($99.4\%$)
      \item \texttt{c256\_56gb}: $8{,}568$\,s of $8{,}589$\,s ($99.8\%$)
    \end{itemize}
    \item Rust emits results without per-cluster dict construction, so it avoids this cost entirely.
    \item We therefore report both columns: suppress-vs-suppress isolates the search engines, with-results-vs-with-results is what a user experiences.
    \item Conflating the two would attribute Python's materialisation cost to Rust's search engineering.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: Why does f16 win nothing standalone?}
  \begin{itemize}
    \item Prediction: halve the footprint, win in the regime where the f32 working set exceeds L3 but the f16 one fits.
    \item On \texttt{c256\_56gb} the per-cluster f32 slice ($\sim$105\,KiB) is already L2-resident. Halving it unlocks no new cache-resident regime, while every comparison pays the f16$\to$f32 dequantisation.
    \item Result: no significant standalone win in the 10-rep fine-$t$ sweep, and a $+15.6\%$ drag inside the bundle at $t{=}16$.
    \item On \texttt{c1024\_56gb} ($3.2\times$ sparser, $\sim$33\,KiB slices, 610 clusters) the aggregate cluster-loop footprint binds. There f16 carries $\sim$20\% of the bundle win.
    \item Footprint optimisations are density-conditional.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: Why ``memory-traffic pressure'' and not ``DRAM bandwidth''?}
  \begin{itemize}
    \item STREAM Triad on this machine reaches $\sim$330\,GB/s aggregate.
    \item Accounting the workload's LLC-miss traffic at $64$\,B per missed probe puts it near $0.5\%$ of that peak.
    \item ``Bandwidth-bound'' claims saturation of the DRAM channels. The counters show nothing close to saturation.
    \item What we observe is anti-scaling from aggregate pressure on the on-socket memory subsystem: queue occupancy and allocator traffic, at utilisation far below peak.
    \item The name matters because it predicts which interventions help: fewer bytes per comparison and per emit. A pure bandwidth upgrade would not.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: The local-ids regression}
  \begin{itemize}
    \item \texttt{local-ids} (per-cluster reindexing) was pre-registered as a positive prediction: narrower IDs, less read-side bandwidth.
    \item Measured: \bad{$+186$ to $+245\%$} at $t \in [8, 64]$ on top of \texttt{packed-ids}. The prediction is falsified.
    \item Mechanism, from a direct \texttt{perf stat} probe: the bw$\approx$13 decoder's access pattern over the packed buffer raises the dTLB miss rate $52\times$ per instruction.
    \item TLB-walk stalls at $\sim$100 cycles each drown the bandwidth saving.
    \item Future work (F1): Transparent Huge Pages (2\,MiB pages) cut TLB-miss frequency $512\times$ per covered page. That probe is the natural follow-up.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: What generalises beyond this machine?}
  \begin{itemize}
    \item The thresholds do not generalise. The methodology does.
    \item All experiments ran on one Sapphire Rapids 8468H configuration. Genoa, Granite Rapids, or Grace would shift per-ceiling thresholds and dispatch boundaries.
    \item The pre-flight procedure (probe the binding ceiling at the target composition) is machine-agnostic. The four-ceiling taxonomy is a hypothesis set to re-test on new hardware, and not a constant.
    \item Density scope: dispatch is validated between $\sim$8{,}200 and $\sim$26{,}300 hists/cluster. Outside that range the policy is a prediction.
    \item A multi-machine deployment would add a fifth candidate ceiling (network) with its own intervention class.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: How was correctness validated?}
  \begin{itemize}
    \item Every Rust build validates bitwise-identical against the Python reference across all workloads: precision $=$ recall $=$ 1.000.
    \item Validation runs inside the benchmark harness, so each measured configuration is also a correctness check. Suppress mode is a stopwatch setting only; correctness is always checked with results on.
    \item Rebinning kernel: verified against the Python reference on 200 queries. Rust rounds half-away, NumPy uses banker's rounding. The maximum cumulative difference is $10^{-4}$ at a handful of rows, with no effect on sort order or query results.
    \item The kernel falls back to the Python pipeline if the extension is not importable or for \texttt{cubic\_spline} bin estimation.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: Why Rust, and not C++, Cython, or Numba?}
  \begin{itemize}
    \item The requirements: no GIL in the hot path, explicit control over layout and allocation, safe shared-memory parallelism, and a maintainable Python integration.
    \item Rust with PyO3 covers all four: Rayon work-stealing without data races, Cargo feature flags that make every ablation a separate reproducible build, and a drop-in extension module behind the existing API.
    \item C++ could reach the same performance. The feature-flag ablation matrix and memory safety under 96-thread parallelism are where Rust paid off in practice.
    \item Numba and Cython remove less of the interpreter overhead and leave allocator and layout decisions with Python. Our measurements show those layers are the actual cost.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: Why does the rebinning speedup drop at 5M histograms?}
  \begin{itemize}
    \item $4.72\times$ (324\,K) and $4.77\times$ (997\,K) drop to $2.18\times$ at 5.02\,M histograms.
    \item Cause: the kernel's own parallelism ceiling. At that scale it runs at $\sim$2 effective cores of 96, with $\sim$46\% of kernel time on a shared per-cluster allocation.
    \item An earlier hypothesis (NumPy bulk operations dominating on the Python side) was tested and refuted.
    \item This is a construction-time cost, paid once per index build. The query-phase findings are unaffected.
  \end{itemize}
\end{frame}
